# drone-cv-detection — Python dependencies
#
# Quick start (CPU inference):
#   python3 -m venv ~/cv_env
#   source ~/cv_env/bin/activate
#   pip install -r requirements.txt
#   pip install -e ".[mqtt]"   # optional: live MQTT monitor support
#
# PyTorch must be installed separately before the rest because the right wheel
# depends on your CUDA version:
#   CPU:   pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
#   CUDA:  pip install torch torchvision  (PyPI default ships CUDA wheels)
#
# The ultralytics and sahi packages declare torch/torchvision as optional deps,
# so pip will install CPU wheels automatically if torch is not yet present.
# For GPU workflows always install torch before everything else.

# ── Core CV ──────────────────────────────────────────────────────────────────

numpy>=1.26.0,<2.1.0

# ── Slicing Aided Hyper Inference (SAHI) ─────────────────────────────────────
